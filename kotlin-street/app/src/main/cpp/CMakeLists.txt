# The on-device language model: llama.cpp plus our small engine (llm_core) and its JNI layer (llm_jni).
#
# Android (the app's externalNativeBuild, arm64-v8a only): llama.cpp as shared libraries, with the CPU kernels built as
# several variants (baseline armv8.0, armv8.2 + dotprod + fp16, armv8.6 + i8mm) of which llm_jni.cpp loads the best one
# the phone supports. So it is fast on modern phones and still runs on any arm64 phone.
#
# Host (tools/llm-host-test.sh): llama.cpp static, tuned for this machine, plus the llm_host_test program.
#
# Where llama.cpp is: -DLLAMA_DIR=... (Gradle passes llama.dir from local.properties), else kotlin-street/third_party/llama.cpp.
# Jolt (physics/) and QuickJS-ng (simjs/) the same way: -DJOLT_DIR / -DQUICKJS_DIR (jolt.dir, quickjs.dir).
cmake_minimum_required(VERSION 3.22)
project(timerpg_llm C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if (NOT LLAMA_DIR)
    set(LLAMA_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/llama.cpp")
endif()
get_filename_component(LLAMA_DIR "${LLAMA_DIR}" ABSOLUTE)
if (NOT EXISTS "${LLAMA_DIR}/include/llama.h" OR NOT EXISTS "${LLAMA_DIR}/CMakeLists.txt")
    message(FATAL_ERROR
        "llama.cpp source not found at ${LLAMA_DIR}.\n"
        "Run kotlin-street/tools/get-llama.sh to fetch it into kotlin-street/third_party/llama.cpp, "
        "or set llama.dir=/path/to/llama.cpp in kotlin-street/local.properties.")
endif()
message(STATUS "llama.cpp: ${LLAMA_DIR}")

# The model must be fast even in debug APKs: always optimise llama.cpp (Gradle also passes CMAKE_BUILD_TYPE=Release).
if (NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(STATUS "CMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}: building llama.cpp with Release flags anyway")
    set(CMAKE_BUILD_TYPE Release)
endif()

macro(force name value)
    set(${name} ${value} CACHE BOOL "" FORCE)
endmacro()

# The libraries and common (its speculative decoding drives the model's MTP head), without curl/openssl; no server,
# tools, examples, tests or web UI.
force(LLAMA_BUILD_COMMON   ON)
force(LLAMA_CURL           OFF)
force(LLAMA_OPENSSL        OFF)
force(LLAMA_BUILD_TESTS    OFF)
force(LLAMA_BUILD_TOOLS    OFF)
force(LLAMA_BUILD_EXAMPLES OFF)
force(LLAMA_BUILD_SERVER   OFF)
force(LLAMA_BUILD_APP      OFF)
force(LLAMA_BUILD_UI       OFF)
# mtmd (llama.cpp's multimodal library) alone, for speech to text (asr_core: Qwen3-ASR)
force(LLAMA_BUILD_MTMD     ON)
force(LLAMA_OPENSSL        OFF)
force(LLAMA_SUBPROCESS     OFF)
force(LLAMA_ALL_WARNINGS   OFF)
force(LLAMA_TOOLS_INSTALL  OFF)
force(GGML_BUILD_TESTS     OFF)
force(GGML_BUILD_EXAMPLES  OFF)
force(GGML_ALL_WARNINGS    OFF)
force(GGML_CCACHE          OFF)
# No GPU or accelerator backends but OpenCL for the phone's Adreno GPU (when its headers and link stub are given:
# -DTIMERPG_OPENCL_HEADERS=... -DTIMERPG_OPENCL_LIB=..., Gradle passes opencl.headers / opencl.lib from local.properties;
# tools/get-opencl.sh makes them). The phone's own libOpenCL.so is used at run time.
foreach (b CUDA HIP MUSA METAL VULKAN OPENCL SYCL WEBGPU HEXAGON RPC BLAS KOMPUTE CANN ZDNN ZENDNN OPENVINO VIRTGPU ET)
    force(GGML_${b} OFF)
endforeach()
set(TIMERPG_OPENCL OFF)
if (ANDROID AND TIMERPG_OPENCL_HEADERS AND TIMERPG_OPENCL_LIB AND EXISTS "${TIMERPG_OPENCL_HEADERS}/CL/cl.h" AND EXISTS "${TIMERPG_OPENCL_LIB}")
    set(TIMERPG_OPENCL ON)
    force(GGML_OPENCL ON)
    force(GGML_OPENCL_USE_ADRENO_KERNELS ON)
    force(GGML_OPENCL_EMBED_KERNELS ON)
    set(OpenCL_INCLUDE_DIR "${TIMERPG_OPENCL_HEADERS}" CACHE PATH "" FORCE)
    set(OpenCL_LIBRARY "${TIMERPG_OPENCL_LIB}" CACHE FILEPATH "" FORCE)
    message(STATUS "OpenCL (Adreno) backend: on")
endif()
# And the Hexagon NPU (llama.cpp's ggml-hexagon) when the Hexagon SDK is given (-DTIMERPG_HEXAGON_SDK=...; Gradle passes
# hexagon.sdk from local.properties; tools/get-hexagon.sh fetches it from Qualcomm's public Snapdragon toolchain image).
# It builds the ARM side (libggml-hexagon.so) and the DSP code for each Hexagon generation (libggml-htp-vNN.so, v79 is the
# S25 Ultra's), which Gradle copies into the APK.
set(TIMERPG_HEXAGON OFF)
if (ANDROID AND TIMERPG_HEXAGON_SDK AND EXISTS "${TIMERPG_HEXAGON_SDK}/build/cmake/hexagon_fun.cmake")
    set(TIMERPG_HEXAGON ON)
    force(GGML_HEXAGON ON)
    set(HEXAGON_SDK_ROOT "${TIMERPG_HEXAGON_SDK}" CACHE PATH "" FORCE)
    file(GLOB _hex_tools LIST_DIRECTORIES true "${TIMERPG_HEXAGON_SDK}/tools/HEXAGON_Tools/*")
    list(GET _hex_tools 0 _hex_tool)
    set(HEXAGON_TOOLS_ROOT "${_hex_tool}" CACHE PATH "" FORCE)
    set(PREBUILT_LIB_DIR "android_aarch64" CACHE STRING "" FORCE)
    message(STATUS "Hexagon NPU backend: on (${HEXAGON_TOOLS_ROOT})")
endif()
# Our patches to llama.cpp (patches/*.patch): patches/llama-*.patch always (the English-first output head: docs/
# ENGINE_TAILORING.md, idea 1), patches/ggml-hexagon-*.patch with the NPU backend (its power vote at run time:
# docs/CONVERSATION.md, "The NPU's power vote"). The llama.cpp tree is never edited (it may be shared by every checkout):
# the build uses a shadow of it in the build directory, every entry a symbolic link to the original except the
# directories the patches touch (each patched file's directory, e.g. src, ggml/src/ggml-hexagon), which are copies with
# the patches applied. Remade only when the patches, the copied directories or the tree's layout change.
set(TIMERPG_LLAMA_BINARY_DIR llama.cpp)
file(GLOB TIMERPG_LLAMA_PATCHES "${CMAKE_CURRENT_LIST_DIR}/patches/llama-*.patch")
file(GLOB TIMERPG_HEXAGON_PATCHES "${CMAKE_CURRENT_LIST_DIR}/patches/ggml-hexagon-*.patch")
list(SORT TIMERPG_LLAMA_PATCHES)
list(SORT TIMERPG_HEXAGON_PATCHES)
set(TIMERPG_PATCHES ${TIMERPG_LLAMA_PATCHES})
if (TIMERPG_HEXAGON)
    list(APPEND TIMERPG_PATCHES ${TIMERPG_HEXAGON_PATCHES})
endif()
if (TIMERPG_PATCHES)
    # (the source shadow apart from the build directory: build outputs must never land in its links, i.e. the original)
    set(_shadow "${CMAKE_BINARY_DIR}/llama.cpp-src-patched")
    # the directories copied: each patched file's own, minus those inside another one
    set(_units "")
    foreach (_p ${TIMERPG_PATCHES})
        file(STRINGS "${_p}" _lines REGEX "^\\+\\+\\+ b/")
        foreach (_l ${_lines})
            string(REGEX REPLACE "^\\+\\+\\+ b/([^\t ]*).*$" "\\1" _f "${_l}")
            get_filename_component(_d "${_f}" DIRECTORY)
            if (_d STREQUAL "")
                message(FATAL_ERROR "${_p} patches ${_f} at the top of llama.cpp: only files in a directory can be shadowed")
            endif()
            list(APPEND _units "${_d}")
        endforeach()
    endforeach()
    list(REMOVE_DUPLICATES _units)
    set(_copies "")
    foreach (_u ${_units})
        set(_inside FALSE)
        foreach (_v ${_units})
            string(FIND "${_u}/" "${_v}/" _at)
            if (NOT _u STREQUAL _v AND _at EQUAL 0)
                set(_inside TRUE)
            endif()
        endforeach()
        if (NOT _inside)
            list(APPEND _copies "${_u}")
        endif()
    endforeach()
    list(SORT _copies)
    # the levels made of real directories (the copies' ancestors), and at each the entry that is not a link
    set(_levels "")
    foreach (_c ${_copies})
        set(_path "${_c}")
        while (NOT _path STREQUAL "")
            get_filename_component(_parent "${_path}" DIRECTORY)
            get_filename_component(_name "${_path}" NAME)
            list(APPEND _levels "/${_parent}")  # ("/" + the path: CMake lists drop empty items)
            string(MAKE_C_IDENTIFIER "lvl_/${_parent}" _var)
            list(APPEND ${_var} "${_name}")
            set(_path "${_parent}")
        endwhile()
    endforeach()
    list(REMOVE_DUPLICATES _levels)
    list(SORT _levels)  # (a parent sorts before its children)
    set(_stamp_text "${LLAMA_DIR}\n")
    foreach (_p ${TIMERPG_PATCHES})
        file(MD5 "${_p}" _h)
        string(APPEND _stamp_text "${_p} ${_h}\n")
    endforeach()
    foreach (_c ${_copies})
        file(GLOB_RECURSE _files RELATIVE "${LLAMA_DIR}/${_c}" "${LLAMA_DIR}/${_c}/*")
        list(SORT _files)
        foreach (_f ${_files})
            file(MD5 "${LLAMA_DIR}/${_c}/${_f}" _h)
            string(APPEND _stamp_text "${_c}/${_f} ${_h}\n")
        endforeach()
    endforeach()
    foreach (_lvl ${_levels})
        file(GLOB _entries RELATIVE "${LLAMA_DIR}${_lvl}" "${LLAMA_DIR}${_lvl}/*" "${LLAMA_DIR}${_lvl}/.*")
        list(SORT _entries)
        string(APPEND _stamp_text "[${_lvl}] ${_entries}\n")
    endforeach()
    set(_stamp "${_shadow}/.timerpg-patched")
    set(_old_stamp "")
    if (EXISTS "${_stamp}")
        file(READ "${_stamp}" _old_stamp)
    endif()
    if (NOT _old_stamp STREQUAL _stamp_text)
        message(STATUS "llama.cpp: making the patched shadow tree ${_shadow} (copies: ${_copies})")
        file(REMOVE_RECURSE "${_shadow}")
        foreach (_lvl ${_levels})
            string(MAKE_C_IDENTIFIER "lvl_${_lvl}" _var)
            set(_keep ${${_var}})
            file(MAKE_DIRECTORY "${_shadow}${_lvl}")
            file(GLOB _entries RELATIVE "${LLAMA_DIR}${_lvl}" "${LLAMA_DIR}${_lvl}/*" "${LLAMA_DIR}${_lvl}/.*")
            foreach (_e ${_entries})
                list(FIND _keep "${_e}" _k)
                if (_k LESS 0 AND NOT _e STREQUAL "." AND NOT _e STREQUAL "..")
                    file(CREATE_LINK "${LLAMA_DIR}${_lvl}/${_e}" "${_shadow}${_lvl}/${_e}" SYMBOLIC)
                endif()
            endforeach()
        endforeach()
        foreach (_c ${_copies})
            get_filename_component(_parent "${_c}" DIRECTORY)
            file(COPY "${LLAMA_DIR}/${_c}" DESTINATION "${_shadow}/${_parent}")
        endforeach()
        find_program(TIMERPG_PATCH_EXE patch)
        find_package(Git QUIET)
        foreach (_p ${TIMERPG_PATCHES})
            if (TIMERPG_PATCH_EXE)
                execute_process(COMMAND "${TIMERPG_PATCH_EXE}" -p1 --forward --batch -i "${_p}"
                                WORKING_DIRECTORY "${_shadow}" RESULT_VARIABLE _rc OUTPUT_VARIABLE _out ERROR_VARIABLE _out)
            elseif (GIT_FOUND)
                execute_process(COMMAND "${GIT_EXECUTABLE}" apply -p1 "${_p}"
                                WORKING_DIRECTORY "${_shadow}" RESULT_VARIABLE _rc OUTPUT_VARIABLE _out ERROR_VARIABLE _out)
            else()
                message(FATAL_ERROR "Neither patch nor git found to apply ${_p}")
            endif()
            if (NOT _rc EQUAL 0)
                file(REMOVE_RECURSE "${_shadow}")
                message(FATAL_ERROR "Could not apply ${_p} to ${LLAMA_DIR} (a llama.cpp version it was not made for?):\n${_out}")
            endif()
            message(STATUS "llama.cpp: applied ${_p}")
        endforeach()
        file(WRITE "${_stamp}" "${_stamp_text}")
    endif()
    set(LLAMA_DIR "${_shadow}")
    # (its own build directory: the HTP skels' ExternalProject caches remember their source directory)
    set(TIMERPG_LLAMA_BINARY_DIR llama.cpp-patched)
    message(STATUS "llama.cpp: patched (${TIMERPG_PATCHES})")
endif()

# ggml's own thread pool (no OpenMP runtime to ship); no llamafile sgemm (not used by our Q4_K model, saves size);
# no KleidiAI (it is downloaded at configure time and mostly speeds up Q4_0).
force(GGML_OPENMP       OFF)
force(GGML_LLAMAFILE    OFF)
force(GGML_CPU_KLEIDIAI OFF)
force(GGML_CPU_REPACK   ON)

if (ANDROID)
    if (NOT ANDROID_ABI STREQUAL "arm64-v8a")
        message(FATAL_ERROR "The language model is built for arm64-v8a only (got ${ANDROID_ABI})")
    endif()
    force(BUILD_SHARED_LIBS     ON)
    force(GGML_NATIVE           OFF)
    force(GGML_BACKEND_DL       ON)
    force(GGML_CPU_ALL_VARIANTS ON)
elseif (LLM_HOST_DL)
    # Host test of the Android-style loading: shared libraries, CPU kernels as separate variant libraries that
    # llm_host_test --cpu-lib loads by name (build the variant target too, e.g. ggml-cpu-haswell).
    force(BUILD_SHARED_LIBS     ON)
    force(GGML_NATIVE           OFF)
    force(GGML_BACKEND_DL       ON)
    force(GGML_CPU_ALL_VARIANTS ON)
else()
    # The host test: one static program, optimised for this machine.
    force(BUILD_SHARED_LIBS     OFF)
    force(GGML_NATIVE           ON)
    force(GGML_BACKEND_DL       OFF)
    force(GGML_CPU_ALL_VARIANTS OFF)
endif()

# EXCLUDE_FROM_ALL: only what our targets need gets built.
add_subdirectory("${LLAMA_DIR}" ${TIMERPG_LLAMA_BINARY_DIR} EXCLUDE_FROM_ALL)

add_library(llm_core STATIC llm_core.cpp guide.cpp asr_core.cpp embed_core.cpp)
target_include_directories(llm_core PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(llm_core PUBLIC llama mtmd llama-common)
set_target_properties(llm_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

if (ANDROID)
    # The CPU kernel variants llm_jni.cpp may load (its list must match). The others ggml defines (SVE, SME) are not
    # built: on phones they are not faster than NEON + i8mm, and armv8.6_1 runs on every phone that has them.
    set(TIMERPG_CPU_VARIANTS ggml-cpu-android_armv8.6_1 ggml-cpu-android_armv8.2_2 ggml-cpu-android_armv8.0_1)
    foreach (v ${TIMERPG_CPU_VARIANTS})
        if (NOT TARGET ${v})
            message(FATAL_ERROR "llama.cpp at ${LLAMA_DIR} has no CPU variant ${v}: use the version tools/get-llama.sh fetches")
        endif()
    endforeach()

    # the game's physics (Jolt: cars now, more later), its own library (docs/VEHICLES.md)
    add_subdirectory(physics)
    # the simulation's JavaScript (QuickJS-ng running assets/sim/sim.js: docs/LOOP.md, "The JavaScript bridge"), its own library
    add_subdirectory(simjs)

    add_library(timerpg_llm SHARED llm_jni.cpp)
    target_link_libraries(timerpg_llm PRIVATE llm_core log)
    # Building the JNI library builds the kernels it loads at run time.
    add_dependencies(timerpg_llm ${TIMERPG_CPU_VARIANTS})
    if (TIMERPG_OPENCL)
        add_dependencies(timerpg_llm ggml-opencl)
    endif()
    if (TIMERPG_HEXAGON)
        # the ARM side, and the DSP code for each Hexagon generation (ExternalProject builds, not built otherwise)
        add_dependencies(timerpg_llm ggml-hexagon)
        foreach (v v73 v75 v79 v81 v85)
            if (TARGET htp-${v})
                add_dependencies(timerpg_llm htp-${v})
            endif()
        endforeach()
    endif()
    # Smaller: drop unused code and do not export anything but the JNI entry points.
    set_target_properties(timerpg_llm PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
    target_link_options(timerpg_llm PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL)
else()
    add_executable(llm_host_test host_test.cpp)
    target_link_libraries(llm_host_test PRIVATE llm_core)
    # nlohmann/json.hpp (llama.cpp's vendored copy) to read the app's dialogue.json
    target_include_directories(llm_host_test PRIVATE "${LLAMA_DIR}/vendor")
    find_package(Threads REQUIRED)
    target_link_libraries(llm_host_test PRIVATE Threads::Threads)
    # speech to text on a computer: tools/asr-host-test.sh
    add_executable(asr_host_test asr_host_test.cpp)
    target_link_libraries(asr_host_test PRIVATE llm_core Threads::Threads)
    # memory search on a computer
    add_executable(embed_host_test embed_host_test.cpp)
    target_link_libraries(embed_host_test PRIVATE llm_core Threads::Threads)
    add_executable(eval_memory eval_memory.cpp)
    target_link_libraries(eval_memory PRIVATE llm_core Threads::Threads)
    target_include_directories(eval_memory PRIVATE "${LLAMA_DIR}/vendor")
endif()
